    7|       |//! This module provides functionality to write ROS1 bag files.

   10|       |//! # ROS1 Bag Format Overview
   11|       |//!
   12|       |//! A ROS1 bag file has the following structure:
   13|       |//! 1. Version line: `#ROSBAG V2.0\n`
   14|       |//! 2. File header record (4096 bytes, padded)
   15|       |//! 3. Chunks containing:
   16|       |//!    - Chunk header
   17|       |//!    - Connection records (metadata for each topic)
   18|       |//!    - Message data records
   19|       |//!    - Index data records (for each connection in the chunk)
   20|       |//! 4. Connection records (summary at end)
   21|       |//! 5. Chunk info records (summary at end)
   22|       |//!
   23|       |//! # Example
   24|       |//!
   25|       |//! ```no_run
   26|       |//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
   27|       |//! use robocodec::io::formats::bag::{BagWriter, BagMessage};
   28|       |//!
   29|       |//! let mut writer = BagWriter::create("output.bag")?;
   30|       |//!
   31|       |//! // Add connections (one per topic)
   32|       |//! let schema = "std_msgs/String#ROS1";
   33|       |//! writer.add_connection(0, "/camera/image", "sensor_msgs/Image", schema)?;
   34|       |//!
   35|       |//! // Write messages
   36|       |//! let timestamp_ns = 1234567890;
   37|       |//! let data = vec![0u8; 10];
   38|       |//! writer.write_message(&BagMessage::new(0, timestamp_ns, data))?;
   39|       |//!
   40|       |//! // Finalize the bag file
