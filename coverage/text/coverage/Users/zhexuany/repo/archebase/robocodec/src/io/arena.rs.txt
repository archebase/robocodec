    5|       |//! Memory-mapped file arena for safe lifetime management.
    6|       |//!
    7|       |//! The `MmapArena` owns the memory-mapped file data and provides safe references
    8|       |//! to its contents. This eliminates the need for unsafe lifetime transmutes by
    9|       |//! establishing clear ownership: the arena owns the data, readers borrow from
   10|       |//! the arena, and iterators borrow from readers.
   11|       |//!
   12|       |//! # Ownership Model
   13|       |//!
   14|       |//! ```text
   15|       |//! MmapArena (owns mmap)
   16|       |//!   ↓
   17|       |//! Reader (borrows from arena for 'arena lifetime)
   18|       |//!   ↓
   19|       |//! Iterator (borrows from reader)
   20|       |//! ```
   21|       |//!
   22|       |//! # Example
   23|       |//!
   24|       |//! ```rust,no_run
   25|       |//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
   26|       |//! use robocodec::io::arena::MmapArena;
   27|       |//!
   28|       |//! // Open a file and create an arena
   29|       |//! let arena = MmapArena::open("data.mcap")?;
   30|       |//!
   31|       |//! // The arena owns the data
   32|       |//! let data: &[u8] = arena.data();
   33|       |//!
   34|       |//! // All references are tied to the arena's lifetime
   35|       |//! // No unsafe transmute needed!
   36|       |//! # Ok(())
   37|       |//! # }
   38|       |//! ```
